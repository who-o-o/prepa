\begin{equation*}
    S_{2n} = \sum_{k=1, impair}^{2n} \frac{(-1)^{k-1}}{k} + \sum_{k=1, pair}^{2n} \frac{(-1)^{k-1}}{k} \\
    &= \sum_{p=1}^{n}\frac{1}{2p-1} - \sum_{p=1}^{n}\frac{1}{2p}\\
    &= \sum_{p=1}^{2n}(\frac{1}{2p-1}+\frac{1}{2p}) - 2\sum_{p=1}^{n}\frac{1}{2p}\\
    &= \sum_{k=1}^{2n}\frac{1}{k} - \sum_{k=1}^{n}\frac{1}{k}\\
    &= \sum_{k=n+1}^{2n}\frac{1}{k}
    \int_{n+1}^{2n+1}\frac{1}{t}dt \leq \sum_{k=n+1}^{2n}\frac{1}{k} \leq \int_{n}^{2n}\frac{1}{t}dt
\end{equation*}
